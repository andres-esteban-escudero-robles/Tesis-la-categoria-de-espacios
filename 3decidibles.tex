\chapter{Objetos decidibles en Topos}

En este capítulo presentaremos la noción de objeto decidible y sus propiedades
más importantes para el resto del presente trabajo. El contexto natural para el
estudio de los objetos decidibles, argumentan Carboni y Janelidze \cite{carboni1996decidable},
es una categoría ``lextensiva'', una categoría con todos los límites finitos y coproductos binarios bien comportados.
En seguida desarrollamos formalmente los conceptos.

\begin{proposition}[label=prop:coprovacio]
  Una categoría con coproductos finitos tiene un objeto inicial.
\end{proposition}
\begin{proof}
  El coproducto vacío satisface la definición de objeto inicial.
\end{proof}

\begin{proposition}[label=prop:amascero]
  Sea $\catC$ una categoría con coproductos finitos, entonces para todo objeto $A$ de $\catC$,
  $A\cong A+0$.
\end{proposition}
\begin{proof}
  Sea $A$ un objeto de $\catC$ y tomemos al coproducto $A+0$. Si $i_1\colon A\to A+0$ es una
  de las inclusiones, veamos que $i_1$ es un isomorfismo. Notemos que podemos construir al
  diagrama
  \begin{equation*}
    \begin{tikzcd}
      A\ar{r}{i_1}\ar[swap]{rd}{\id_A}\ar[bend right,swap]{rdd}{i_1} & A+0\ar{d}{\phi} & 0\ar[swap]{l}{!}\ar{ld}{!}\ar[bend left]{ldd}{!} \\
      & A\ar{d}{i_1} \\
      & A+0
    \end{tikzcd}
  \end{equation*}
  donde $\phi$ esta inducida por la propiedad universal, así que cada parte conmuta. Por
  construcción, $\phi i_1 = \id_A$ y como podemos poner a $\id_{A+0}$ en lugar de $i_1\phi$ en el diagrama,
  concluimos que $i_1\phi=\id_{A+0}$.
\end{proof}

\begin{proposition}[label=prop:inclusionesidenticas]
  Si $\catC$ tiene coproductos finitos y $B\to B\leftarrow B$ es un coproducto 
  con inclusiones idénticas, entonces para cualquier $C\in\catC$, hay a lo más un morfismo $B\to C$.
\end{proposition}
\begin{proof}
  Supongamos que tenemos $f,g\colon B\to C$ morfismos y que $i\colon B\to B$ es la inclusión, 
  entonces $f=(f+g)i=g$. 
\end{proof}

\begin{definition}
  Una categoría $\catC$ con coproductos es extensiva si para cada par de objetos $A,B$
  el funtor canónico
  \begin{equation*}
    +\colon\catC/A\times\catC/B \to \catC/(A+B)
  \end{equation*}
  es un equivalencia de categorías.
\end{definition}

El siguiente teorema nos da una definición equivalente de categoría extensiva que
resulta muy útil en la práctica. La demostración se puede encontrar en \cite{carboni1993introduction}.

\begin{theorem}[label=thm:condextensiva]
  Una categoría $\catC$ con coproductos es extensiva si y solo si tiene productos
  fibrados a lo largo de inclusiones de coproductos y para cada diagrama conmutativo
  \begin{equation*}
    \begin{tikzcd}
      A_1\ar{r}{a_1}\ar[swap]{d}{f_1} & A\ar{d}{f} & A_2\ar[swap]{l}{a_2}\ar{d}{f_2} \\
      X\ar[swap]{r}{x} & X+Y & Y\ar{l}{y}
    \end{tikzcd}
  \end{equation*}
  la fila superior es un coproducto si y solo si ambos cuadrados son productos fibrados.
\end{theorem}
 
Se dice que los coproductos en una categoría extensiva son bien comportados,
ya que se comportan de manera similar a los coproductos en $\con$. Para justificar
esta afirmación, empezamos con la siguiente definición.

\begin{definition}
  En una categoría con coproductos y productos fibrados a lo largo de inclusiones, 
  diremos que un coproducto binario es disjunto si el producto fibrado de las inclusiones
  es el objeto inicial y las inclusiones son monomorfismos. 
\end{definition}

\begin{proposition}
  Si $\catC$ es extensiva, entonces los coproductos son disjuntos.
\end{proposition}
\begin{proof}
  Sea $\catC$ una categoría extensiva y $A,B\in\catC$, consideramos al diagrama conmutativo
  \begin{equation*}
    \begin{tikzcd}
      0\ar{r}{}\ar[swap]{d}{} & B\ar{d}{b} & B\ar[swap]{l}{\id}\ar{d}{\id} \\
      A\ar[swap]{r}{a} & A+B & B.\ar{l}{b}
    \end{tikzcd}
  \end{equation*}
La parte superior es un coproducto por~\ref{prop:amascero}, entonces el teorema~\ref{thm:condextensiva} dice que ambos cuadrados 
son productos fibrados, el izquierdo dice que el producto fibrado de $a$ y $b$ es el objeto inicial
y el derecho implica que $b$ es un monomorfismo. Pero análogamente se puede demostrar que
$a$ es un monomorfismo, por lo que los coproductos son disjuntos.
\end{proof}

\begin{definition}
  Sea $\catC$ una categoría con coproductos y productos fibrados a lo largo de inclusiones.
  Los coproductos son universales si calcular los productos fibrados a lo largo de inclusiones
  resulta en un diagrama de coproducto.
\end{definition}

\begin{corollary}
  Si $\catC$ es extensiva, entonces los coproductos son universales.
\end{corollary}
\begin{proof}
  Es inmediato gracias al teorema~\ref{thm:condextensiva}.
\end{proof}

\begin{lemma}
  En una categoría extensiva, el objeto inicial es estricto, es decir, si existe $f\colon A\to 0$ entonces $A\cong 0$.
\end{lemma}
\begin{proof}
  Sea $\catC$ una categoría extensiva y supongamos que existe un morfismo $f\colon A\to 0$.
  Consideremos al diagrama
  \begin{equation*}
    \begin{tikzcd}
      A\ar{r}{\id}\ar[swap]{d}{f} & A\ar{d}{f} & A\ar[swap]{l}{\id}\ar{d}{f} \\
      0\ar[swap]{r}{\id} & 0+0 & 0\ar{l}{\id}
    \end{tikzcd}
  \end{equation*}
  que conmuta trivialmente, más aún, es fácil demostrar que un cuadrado formado por la misma flecha e identidades siempre es un
  producto fibrado. Entonces, aplicando el teorema~\ref{thm:condextensiva}, la fila superior es un coproducto.
  Ahora, para demostrar que $A$ sea inicial, para cualquier objeto $B$ debe existir un único morfismo $A\to B$. 
  La unicidad se da por~\ref{prop:inclusionesidenticas}, mientras que $!_Bf\colon A\to B$ existe por hipótesis. 
  Por lo tanto, $A\cong 0$.
\end{proof}

\begin{theorem}
  Dada una categoría $\catC$ con coproductos finitos, $\catC$ es extensiva si y solo si sus coproductos
  son disjuntos y universales.
\end{theorem}
\begin{proof}
  Solo falta demostrar el regreso. Sea $\catC$ una categoría con coproductos
  disjuntos y universales, queremos demostrar que dado un diagrama conmutativo
  de la forma:
  \begin{equation*}
    \begin{tikzcd}
      A_1\ar{r}{a_1}\ar[swap]{d}{f_1} & A\ar{d}{f} & A_2\ar[swap]{l}{a_2}\ar{d}{f_2} \\
      X_1\ar[swap]{r}{x_1} & X_1+X_2 & X_2,\ar{l}{x_2}
    \end{tikzcd}
  \end{equation*}
  la fila superior es un coproducto si y solo si ambos cuadrados son productos fibrados.
  Por la condición de universalidad el regreso es inmediato, entonces suponemos
  que la fila superior es un coproducto. Ahora tomamos $g\colon B\to X_1$, $h\colon B\to A$ tales que $x_1g = fh$
  y calculamos los productos fibrados a lo largo de $a_1$ y $a_2$, lo cual nos da al diagrama conmutativo:
  \begin{equation}\label{eq:ceroestricto}
    \begin{tikzcd}
      B_1\ar{r}{b_1}\ar[swap]{d}{h_1}\ar[phantom, pos=0.1]{rd}{\lrcorner} & B\ar{d}{h} & B_2\ar[swap]{l}{b_2}\ar{d}{h_2}\ar[phantom, pos=0.1]{ld}{\llcorner} \\
      A_1\ar[tail,swap]{r}{a_1}\ar[swap]{d}{f_1} & A_1+A_2\ar{d}{f} & A_2\ar[tail]{l}{a_2}\ar{d}{f_2} \\
      X_1\ar[tail,swap]{r}{x_1} & X_1+X_2 & X_2.\ar[tail]{l}{x_2}
    \end{tikzcd}
  \end{equation}
  Por la universalidad de los coproductos, $B\cong B_1+B_2$ y si pasara que $B_2\cong 0$,
  $b_1$ sería un isomorfismo con lo que $h_1b_1^{-1}$ sería un buen candidato al único morfismo
  $B\to A_1$ para mostrar que $A_1$ es un producto fibrado. Como los coproductos son disjuntos,
  el producto fibrado de $x_1$ con $x_2$ es $0$, así que consideramos al diagrama
  \begin{equation*}
    \begin{tikzcd}
      B_2\ar{r}{h_2}\ar[swap]{d}{b_2} & A_2\ar{d}{a_2}\ar{rd}{f_2} & {} \\
      B\ar{r}{h}\ar[swap]{rd}{g} & A_1+A_2\ar{rd}{f} & X_2\ar{d}{x_2} \\
      {} & X_1\ar[swap]{r}{x_1} & X_1+X_2,
    \end{tikzcd}
  \end{equation*}
  donde cada parte conmuta por cómo pedimos a $B$ y por \eqref{eq:ceroestricto}, por lo que
  el exterior conmuta y, al ser $0$ estricto, $B_2\cong 0$. Entonces proponemos
  a $h_1b_1^{-1}$ como el único morfismo de $B$ a $A_1$ que hace que el siguiente diagrama
  sea un producto fibrado:
  \begin{equation*}
    \begin{tikzcd}
      B\ar[bend left]{drr}{h}\ar[bend right,swap]{ddr}{g}\ar{dr}{h_1b_1^{-1}} \\
      & A_1\ar{r}{a_1}\ar[swap]{d}{f_1} & A_1+A_2\ar{d}{f} \\
      & X_1\ar[swap]{r}{x_1} & X_1+X_2.
    \end{tikzcd}
  \end{equation*}
  Sabemos que $x_1$ y $a_1$ son mono porque los coproductos son disjuntos, de manera que,
  el lector puede verificar la conmutatividad de los triángulos persiguiendo al diagrama
  \eqref{eq:ceroestricto} y usando que $x_1$ es mono. Por otro lado, que $a_1$ sea mono
  nos da la unicidad de $h_1b_1^{-1}$, así que $A_1$ es un producto fibrado. Un argumento 
  simétrico demuestra que $A_2$ también es un producto fibrado.
\end{proof}

Para el resto del capítulo nos será de utilidad contar con límites finitos en la categoría ambiente.

\begin{definition}
  Una categoría $\catC$ es ``lextensiva'' si es finitamente completa (es una categoría lex)
  y es extensiva.
\end{definition}

Por ``objeto decidible'' en una categoría queremos referirnos a aquellos objetos cuyos elementos son distinguibles entre sí,
desde el lenguaje de la categoría ambiente. Por ejemplo, de manera externa, dos puntos de un espacio topológico
pueden ser distintos, pero no separables, haciendo que el espacio no los pueda distinguir. Entonces, necesitamos
una categoría que nos permita expresar que el predicado de igualdad cumple la ley del tercero excluido, es decir,
la diagonal de un objeto tiene complemento. Ahora mostramos que las categorías lextesivas soportan una noción adecuada de objeto decidible. 

\begin{proposition}
  Sea $\catC$ una categoría lextensiva y $A\in\catC$, el orden parcial $\sub(A)$
  tiene $0$, $1$ e ínfimos.
\end{proposition}
\begin{proof}
  Como el $0$ de $\catC$ es estricto, automáticamente toda flecha que sale del inicial es mono,
  entonces $0\to A$ es el mínimo de $\sub(A)$, mientras que la identidad es el elemento máximo.
  El ínfimo de dos subobjetos de $A$ es su producto fibrado.
\end{proof}

\begin{proposition}
  En una categoría $\catC$ lextensiva, para cada $f\colon A\to B$ podemos construir al
  funtor producto fibrado $f^*\colon\catC/B\to\catC/A$ y éste funtor preserva coproductos. 
\end{proposition}
\begin{proof}
  Sea $f\colon A\to B$ un morfismo de $\catC$ fijo, definimos a $f^*$ en una rebanada
  \begin{equation*}
    \begin{tikzcd}
      S\ar{r}{g}\ar[swap]{d}{s} & T\ar{ld}{t} \\
      B
    \end{tikzcd}
  \end{equation*}
  calculando los productos fibrados con $f$:
  \begin{equation*}
    \begin{tikzcd}
      S'\ar{rr}{p}\ar[dashed]{rd}{f^*(g)}\ar[swap]{dd}{f^*S} & {} & S\ar{rd}{g}\ar[near start]{dd}{s} \\
      {} & T'\ar[near start]{rr}{q}\ar{ld}{f^*T} & {} & T\ar{ld}{t} \\
      A\ar[swap]{rr}{f} & {} & B.
    \end{tikzcd}
  \end{equation*}
  Hacemos notar que dos caras del prisma son los productos fibrados que calculamos
  y el lema del producto fibrado implica que el tercer cuadrado también es un producto fibrado.
  De esta manera, si $h\colon S\to S+T$, $k\colon T\to S+T$ son inclusiones, $s+t$ es 
  el coproducto en $\catC/B$ y tenemos al diagrama
  \begin{equation*}
    \begin{tikzcd}
      S'\ar{r}{f^*(h)}\ar[swap]{d}{p} & (S+T)'\ar{d}{r} & T'\ar[swap]{l}{f^*(k)}\ar{d}{q} \\
      S\ar[swap]{r}{h} & S+T & T,\ar{l}{k}
    \end{tikzcd}
  \end{equation*}
  donde ambos cuadros son productos fibrados, así que, la extensionalidad nos da que
  la fila superior del diagrama es un coproducto en $\catC$, pero uno puedo verificar fácilmente
  que $f^*(S+T)\colon(S+T)'\to A$ es el coproducto $f^*S+f^*T$ en $\catC/A$. 
\end{proof}

\begin{corollary}
  En una categoría lextensiva $\catC$, si $A\in\catC$ y $s\colon S\to A$, $t\colon T\to A$
  con subobjetos disjuntos de $A$ (es decir, $S\land T\cong 0$), entonces $S+T$ es su supremo en $\sub(A)$.
\end{corollary}
\begin{proof}
  Sea $i_1\colon S\to S+T\leftarrow T\noloc i_2$ el coproducto de $S$ y $T$, la flecha $h+k$ es el coproducto de $h$ y $k$ en $\catC/A$, entonces $k^*(h+k)\cong k^*h+k^*k$.
  Dado que $k$ es mono, $k^*k=\id_T$ y por hipótesis $k^*h=0$, así que $k^*(h+k)\cong\id_T$:
  \begin{equation*}
    \begin{tikzcd}
      T\ar{r}{i_2}\ar[swap]{d}{\id}\ar[phantom, pos=0.1]{rd}{\lrcorner} & S+T\ar{d}{h+k} \\
      T\ar[swap]{r}{k} & A.
    \end{tikzcd}
  \end{equation*}
  Por simetría, el diagrama dice que $(h+k)^*k=i_2$ y un argumento análogo muestra
  que $(h+k)^*h=i_1$. Como $(h+k)^*(h+k)\cong(h+k)^*h+(h+k)^*k$, por lo anterior obtenemos que $(h+k)^*(h+k)=h+k$,
  así que $h+k\colon S+T\to A$ es mono y claramente es el supremo de $S$ y $T$ en $\sub(A)$.
\end{proof}

Con el resultado anterior, vemos que, en una categoría lextensiva, tiene sentido
preguntarse por subobjetos complementados.

\begin{definition}
  Sea $\catC$ una categoría y $A\in\catC$. Si $\sub(A)$ es una retícula, decimos que
  un subobjeto $S$ es complementado si existe un subobjeto $\overline{S}$ tal que
  $S\land\overline{S}=0$ y $S\lor\overline{S}=A$.
\end{definition}

\begin{definition}
  En una categoría lextensiva $\catC$ un objeto $D$ es decidible si para su diagonal, 
  $\delta=\id_D\times\id_D$, existe una flecha $\overline{\delta}\colon \overline{D}\to D\times D$ 
  tal que el diagrama
  \begin{equation*}
    \begin{tikzcd}
      D\ar{r}{\delta} & D\times D & \overline{D}\ar[swap]{l}{\overline{\delta}}
    \end{tikzcd}
  \end{equation*}
  es un coproducto.
\end{definition}

\begin{proposition}
  Sea $\catC$ una categoría lextensiva. Si $D$ es un objeto decidible, $\delta\colon D\to D\times D$ es un subobjeto complementado
  de $D\times D$.
\end{proposition}
\begin{proof}
  Proponemos a $\overline{\delta}\colon\overline{D}\to D\times D$ como el complemento deseado.
  Como los coproductos son disjuntos, $D\land\overline{D}=0$ y con ello sucede
  que $D\lor\overline{D}=D+\overline{D}\cong D\times\overline{D}$.
\end{proof}

Así que, si una categoría lextensiva tiene una lógica interna, nuestra definición
de objeto decidible sí captura la idea de que la igualdad en el objeto cumpla la ley
del tercero excluido. 

En los siguientes resultados, asumimos que estamos trabajando en una categoría lextensiva.

\begin{proposition}
  Los subobjetos de un objeto decidible son decidibles.
\end{proposition}
\begin{proof}
  Sea $A$ un objeto decidible con $m\colon S\to A$ un subobjeto de $A$, notamos que
  $m$ induce $m\times m\colon S\times S \rightarrowtail A\times A$ y no es difícil 
  demostrar que el siguiente es un producto fibrado:
  \begin{equation*}
    \begin{tikzcd}
        S\ar{r}{m} \ar[tail]{d}[swap]{\delta_S} & A\ar[tail]{d}{\delta_A}\\
        S\times S\ar{r}[swap]{m\times m} & A\times A.
    \end{tikzcd}
  \end{equation*}
  Si además calculamos el producto fibrado de $\overline{\delta}_A$, obtenemos
  \begin{equation*}
    \begin{tikzcd}
        S\ar[swap]{d}{m}\ar{r}{\delta_S} & S\times S\ar{d}{m\times m} & S'\ar[phantom,pos=0.1]{ld}{\llcorner}\ar[swap]{l}{p}\ar{d}{q} \\
        A\ar[swap]{r}{\delta_A} & A\times A & \overline{A}\ar{l}{\overline{\delta}_A}
    \end{tikzcd}
  \end{equation*}
  y por extensividad, la fila superior es un diagrama de coproducto.
\end{proof}

\begin{corollary}[label=prop:cerounodiscreto]
  El objeto terminal e inicial son objetos decidibles.
\end{corollary}
\begin{proof}
  El $1$ es decidible puesto que el diagrama $1\to1\times1\leftarrow0$ es un coproducto
  si recordamos que $1\times1\cong1\cong1+0$. Por su parte, el $0$ es subobjeto de todos, 
  en particular del $1$.
\end{proof}

\begin{theorem}[label=thm:onndecidible]
  Sea $\mathcal{E}$ un topos con $(N,o,s)$ un objeto de números naturales, entonces $N$ es un objeto decidible.
\end{theorem}
\begin{proof}
  El plan es ver a $\delta\colon N\to N\times N$ como intersección de subobjetos complementados en $\sub(N\times N)$ (así va la demostración de Johnstone pero hay una más directa con la lógica interna).
  Primero, en un topos tenemos recursividad con parámetros que nos permite definir a la resta truncada como 
  $\dot{-}\colon N\times N\to N$ tal que $\dot{-}(a,0)=a$ y $\dot{-}(a,n+1)=p(\dot{-}(a,n))$ donde $p$ es el predecesor de~\ref{def:predecesor}.
  Ahora construimos al subobjeto:
  \begin{equation*}
    \begin{tikzcd}
      N_1\ar{r}{}\ar[tail]{d}{} & 1\ar{d}{o} \\
      N\times N\ar{r}{\dot{-}} & N,
    \end{tikzcd}
  \end{equation*}
  que es complementado gracias a que ser complementado es estable bajo productos fibrados. Como $n\dot{-}n=0$ para toda $n$,
  $\delta$ es subobjeto de $N_1$ y por lo tanto $\delta$ también tiene complemento en $\sub(N\times N)$.
\end{proof}

\begin{proposition}
  Si un topos $\toposE$ cumple el \textit{Nullstellensatz} y $B$ es un objeto decidible, entonces para $f,g\colon A\to B$
  con $f\neq g$ existe un elemento global $p\colon 1\to A$ tal que $f p \neq g p$.
\end{proposition}
\begin{proof}
  Sea $\toposE$ un topos que satisface el \textit{Nullstellensatz}, $B$ un objeto decidible en $\toposE$ y $f,g\colon A\to B$ con $f\neq g$. Consideramos al igualador $e\colon E\to A$ de $f$ y $g$, 
  quien de hecho es el producto fibrado entre $\delta\colon B\to B\times B$ y $\langle f,g\rangle\colon A\to B\times B$.
  Dado que $B$ es complementado en $\sub(B\times B)$, $E$ es complementado en $\sub(A)$, llamemos $\overline{e}\colon \overline{E}\to A$ a su complemento.
  Como $f\neq g$, $E\not\cong A$ y por lo tanto $\overline{E}\not\cong 0$. Por el \textit{Nullstellensatz}, $\overline{E}$ tiene un elemento global $p\colon 1\to \overline{E}$ y por lo tanto $e p\colon 1\to A$ es un elemento global de $A$ tal que $f e p \neq g e p$.
\end{proof}
